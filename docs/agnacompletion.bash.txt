# agna CLI completion for bash
# source this file from ~/.bashrc, e.g.:
#   [ -f /usr/local/share/agna/agnacompletion.bash ] &&
#       . /usr/local/share/agna/agnacompletion.bash
#
# Completes commands, metric names, analyses, transform ops, layouts,
# formats, types and option values, with file completion as fallback.

_agna()
{
    local cur prev word
    cur="${COMP_WORDS[COMP_CWORD]}"
    prev="${COMP_WORDS[COMP_CWORD-1]}"

    local commands="version info analyse convert transform draw generate matrix nodes ego components metrics distance diff add-scalar multiply-scalar transpose symmetrize normalize square merge-networks remove-outsiders delete-nodes add-nodes renumber from-chain layout set --help --version"
    local analyses="basic density cohesion nodal indegree outdegree emission reception determination status geodesics open-chain eccentricity diameter bavelas closeness fareness betweenness prestige cliques full"
    local metric_names="density diameter eccentricity closeness betweenness indegree outdegree total-degree emission reception status determination geodesics"
    local transform_ops="transpose symmetrize-max symmetrize-sum symmetrize-below normalize-binary normalize-threshold: add-scalar: multiply-scalar: square merge: delete-node: delete-nodes: isolate: merge-nodes: remove-outsiders renumber add-nodes: clone-node:"
    local layouts="circular random spring grid concentric"
    local layout_cmds="circular random spring grid concentric star"
    local formats="agn txt csv net graphml gml graphson xls"
    local types="random star circular"
    local sym_modes="below sum max"
    local on_off="on off"
    local set_flags="--name --names-visible --names-x --names-y --title-visible --title-x --title-y --grid-visible --grid-step --separator --grid-transparency --max-transparency --edge-value-visible --edge-value-position --edge-value-color --edge-color --names-color --grid-color --title-color --background-color --background-image --background-image-x --background-image-y --background-image-width --background-image-height --faces-visible --allow-edge-selection --color-fidelity --snap-to-grid --default-face"

    # value completion for the previous token
    case "$prev" in
        --layout) COMPREPLY=( $(compgen -W "$layout_cmds" -- "$cur") ); return ;;
        --type)   COMPREPLY=( $(compgen -W "$types" -- "$cur") ); return ;;
        --format) COMPREPLY=( $(compgen -W "csv tsv json" -- "$cur") ); return ;;
        --in-format|--out-format) COMPREPLY=( $(compgen -W "$formats" -- "$cur") ); return ;;
        --op)     COMPREPLY=( $(compgen -W "$transform_ops" -- "$cur") ); return ;;
        --mode)   COMPREPLY=( $(compgen -W "$sym_modes" -- "$cur") ); return ;;
        --names-visible|--title-visible|--grid-visible|--edge-value-visible|--faces-visible|--allow-edge-selection|--color-fidelity|--snap-to-grid)
                  COMPREPLY=( $(compgen -W "$on_off" -- "$cur") ); return ;;
        --background|--edge-value-color|--edge-color|--names-color|--grid-color|--title-color|--background-color)
                  COMPREPLY=( $(compgen -W "#" -- "$cur") ); return ;;
        --out|--background-image|--default-face) COMPREPLY=( $(compgen -f -- "$cur") ); return ;;
    esac

    if [[ $COMP_CWORD -eq 1 ]]; then
        COMPREPLY=( $(compgen -W "$commands" -- "$cur") )
        return
    fi

    local cmd="${COMP_WORDS[1]}"
    case "$cmd" in
        analyse)
            COMPREPLY=( $(compgen -W "$analyses --all --out" -- "$cur") )
            ;;
        metrics)
            COMPREPLY=( $(compgen -W "$metric_names --all --format --out" -- "$cur") )
            ;;
        transform)
            COMPREPLY=( $(compgen -W "$transform_ops --in-format --out-format" -- "$cur") )
            ;;
        draw)
            COMPREPLY=( $(compgen -W "--layout --size --labels --background --no-faces --edge-values --out" -- "$cur") )
            ;;
        layout)
            COMPREPLY=( $(compgen -W "--layout --size --out --out-format" -- "$cur") )
            ;;
        generate)
            COMPREPLY=( $(compgen -W "--nodes --out --type --seed --degree" -- "$cur") )
            ;;
        convert)
            COMPREPLY=( $(compgen -W "--in-format --out-format" -- "$cur") )
            ;;
        ego|distance)
            COMPREPLY=( $(compgen -W "--node --from --to --out" -- "$cur") )
            ;;
        add-scalar|multiply-scalar)
            COMPREPLY=( $(compgen -W "--out --out-format" -- "$cur") )
            ;;
        symmetrize)
            COMPREPLY=( $(compgen -W "--mode --out --out-format" -- "$cur") )
            ;;
        normalize)
            COMPREPLY=( $(compgen -W "--binary --threshold --out --out-format" -- "$cur") )
            ;;
        merge-networks)
            COMPREPLY=( $(compgen -W "--mode --out --out-format" -- "$cur") )
            ;;
        add-nodes)
            COMPREPLY=( $(compgen -W "--count --out --out-format" -- "$cur") )
            ;;
        set)
            COMPREPLY=( $(compgen -W "$set_flags --out" -- "$cur") )
            ;;
        from-chain|transpose|square|remove-outsiders|delete-nodes|renumber)
            COMPREPLY=( $(compgen -W "--out --out-format" -- "$cur") )
            ;;
        *)  # FILE arguments everywhere else
            COMPREPLY=( $(compgen -f -- "$cur") )
            ;;
    esac
    return
}

complete -o default -F _agna agna
complete -o default -F _agna java
