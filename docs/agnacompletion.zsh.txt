#compdef agna
# agna CLI completion for zsh
# place this file on your fpath as _agna, e.g.:
#   mkdir -p ~/.zfunc && cp docs/agnacompletion.zsh ~/.zfunc/_agna
#   printf 'fpath=(~/.zfunc $fpath)\nautoload -Uz compinit && compinit\n' >> ~/.zshrc
#
# Completes commands, metric names, analyses, transform ops, layouts,
# formats, types, option values and file arguments.

_agna() {
  local -a commands analyses metric_names transform_ops layouts formats types modes bools
  commands=(
    'version:print the version'
    'info:network summary'
    'analyse:run analyses (--all, names, cliques:N, --out)'
    'convert:convert formats'
    'transform:apply an operation and save as a new network'
    'draw:render the network to PNG'
    'generate:generate a synthetic network'
    'matrix:print the adjacency matrix'
    'nodes:print the node table'
    'ego:extract a 1-hop ego network'
    'components:list connected components'
    'metrics:structured metrics (CSV/JSON)'
    'distance:shortest path between two nodes'
    'diff:structural comparison'
    'add-scalar:add a scalar to every cell'
    'multiply-scalar:multiply every cell by a scalar'
    'transpose:transpose the sociomatrix'
    'symmetrize:symmetrize the matrix'
    'normalize:binarize the matrix'
    'square:matrix square'
    'merge-networks:merge two networks'
    'remove-outsiders:delete isolated nodes'
    'delete-nodes:delete named nodes'
    'add-nodes:append new nodes'
    'renumber:name nodes 1..n'
    'from-chain:create a network from a chain file'
    'layout:apply coordinates without rendering'
    'set:edit viewer attributes'
    '--help:this text'
    '--version:print the version'
  )
  analyses=(basic density cohesion nodal indegree outdegree emission reception determination status geodesics open-chain eccentricity diameter bavelas closeness fareness betweenness prestige cliques full)
  metric_names=(density diameter eccentricity closeness betweenness indegree outdegree total-degree emission reception status determination geodesics)
  transform_ops=(transpose symmetrize-max symmetrize-sum symmetrize-below normalize-binary normalize-threshold: add-scalar: multiply-scalar: square merge: delete-node: delete-nodes: isolate: merge-nodes: remove-outsiders renumber add-nodes: clone-node:)
  layouts=(circular random spring grid concentric star)
  formats=(agn txt csv net graphml gml graphson xls)
  types=(random star circular)
  modes=(below sum max)
  bools=(on off)

  local -a opts
  case $words[2] in
    analyse)   opts=( $analyses '--all' '--out' ) ;;
    metrics)   opts=( $metric_names '--all' '--format' '--out' ) ;;
    transform) opts=( $transform_ops '--in-format' '--out-format' ) ;;
    draw)      opts=( '--layout' '--size' '--labels' '--background' '--no-faces' '--edge-values' '--out' ) ;;
    generate)  opts=( '--nodes' '--out' '--type' '--seed' '--degree' ) ;;
    convert)   opts=( '--in-format' '--out-format' ) ;;
    ego)       opts=( '--node' '--out' ) ;;
    distance)  opts=( '--from' '--to' ) ;;
    diff)      opts=( '--out' ) ;;
    add-scalar|multiply-scalar|from-chain|transpose|square|remove-outsiders|delete-nodes|renumber) opts=( '--out' '--out-format' ) ;;
    symmetrize|merge-networks) opts=( '--mode' '--out' '--out-format' ) ;;
    normalize) opts=( '--binary' '--threshold' '--out' '--out-format' ) ;;
    add-nodes) opts=( '--count' '--out' '--out-format' ) ;;
    layout)    opts=( '--layout' '--size' '--out' '--out-format' ) ;;
    set)       opts=( '--name' '--names-visible' '--names-x' '--names-y' '--title-visible' '--title-x' '--title-y' '--grid-visible' '--grid-step' '--separator' '--grid-transparency' '--max-transparency' '--edge-value-visible' '--edge-value-position' '--edge-value-color' '--edge-color' '--names-color' '--grid-color' '--title-color' '--background-color' '--background-image' '--background-image-x' '--background-image-y' '--background-image-width' '--background-image-height' '--faces-visible' '--allow-edge-selection' '--color-fidelity' '--snap-to-grid' '--default-face' '--out' ) ;;
    *)         _describe 'command' commands; return ;;
  esac

  case $words[CURRENT-1] in
    --layout) compadd -- $layouts; return ;;
    --type)   compadd -- $types; return ;;
    --format) compadd -- csv tsv json; return ;;
    --in-format|--out-format) compadd -- $formats; return ;;
    --op)     compadd -- $transform_ops; return ;;
    --mode)   compadd -- $modes; return ;;
    --names-visible|--title-visible|--grid-visible|--edge-value-visible|--faces-visible|--allow-edge-selection|--color-fidelity|--snap-to-grid) compadd -- $bools; return ;;
    --out|--node|--from|--to|--background-image|--default-face) _files; return ;;
    --background|--edge-value-color|--edge-color|--names-color|--grid-color|--title-color|--background-color) compadd -- '#'; return ;;
    *)        compadd -- $opts; _files ;;
  esac
}

_agna "$@"
